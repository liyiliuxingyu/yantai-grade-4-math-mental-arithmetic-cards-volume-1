\begingroup\color{answerred}\noindent（1）$a\perp b$\par
\noindent（2）$c\perp d,\quad d\perp e$\par
\noindent（3）$b\perp d$\par\par\endgroup
\keyheading{第3题　直线的位置关系}
\begingroup\color{answerred}平行：①⑥；垂直：②⑦⑧。\par 相交：②③④⑤⑦⑧。\par \textbf{分类口径：}若三个框不重复填序号，中间“相交”框填③④⑤（相交但不垂直）。\par\endgroup
